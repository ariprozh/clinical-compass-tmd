Раздельные шкалы и план безопасности

Разные домены боли, функции челюсти и эмоционального состояния не складываются в единый индекс. При повышенных баллах PHQ-4 срабатывает протокол информирования участника.

Независимые единицы измерения

GCPS (CPI и влияние) рассчитывается в шкале 0–100; JFLS-8 — средний балл 0–10; PHQ-4 — сумма 0–12 с выделением подшкал тревоги (0–6) и депрессии (0–6).

Маршрутизация при высоком PHQ-4

При балле депрессии ≥ 3 или общем балле PHQ-4 ≥ 6 пациенту отображается деликатная рекомендация обсудить состояние с лечащим врачом и телефоны поддержки.

// Google Apps Script для приёма данных протокола DC/TMD Axis II («Клинический Компас»)
const SHEET_NAME = 'DC_TMD_AxisII_Data';

function getOrCreateSheet() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sheet = ss.getSheetByName(SHEET_NAME);
  if (!sheet) {
    sheet = ss.insertSheet(SHEET_NAME);
    const headers = [
      'record_uuid', 'assessment_datetime', 'participant_id', 'visit_id',
      'consent_status', 'consent_timestamp', 'age_18_plus', 'acute_trauma_exclusion',
      'gcps_now', 'gcps_worst', 'gcps_avg', 'gcps_cpi_0_100',
      'gcps_interf_daily', 'gcps_interf_social', 'gcps_interf_work', 'gcps_interf_mean_0_100',
      'pain_drawing_count', 'pain_drawing_zones',
      'jfls8_mean_0_10', 'jfls8_answered_count', 'jfls8_items_json',
      'phq4_total_0_12', 'phq4_anxiety_0_6', 'phq4_depression_0_6', 'safety_flag',
      'treatment_change', 'treatment_notes', 'missing_status', 'instrument_version'
    ];
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight('bold').setBackground('#e0effe');
    sheet.setFrozenRows(1);
  }
  return sheet;
}

function doGet(e) {
  getOrCreateSheet();
  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    message: 'Клинический Компас: Google Apps Script активен и готов к приёму анкет DC/TMD Axis II.'
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const sheet = getOrCreateSheet();
    const raw = e.postData && e.postData.contents ? e.postData.contents : '{}';
    const data = JSON.parse(raw);

    if (data.action === 'ping') {
      return ContentService.createTextOutput(JSON.stringify({
        status: 'ok',
        message: 'Тестовое соединение с Google Таблицей успешно установлено!'
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const row = [
      data.record_uuid || Utilities.getUuid(),
      data.assessment_datetime || new Date().toISOString(),
      data.participant_id || '',
      data.visit_id || '',
      data.consent_status || '',
      data.consent_timestamp || '',
      data.age_18_plus ? 'YES' : 'NO',
      data.acute_trauma_exclusion || '',
      data.gcps?.pain_now ?? '',
      data.gcps?.pain_worst ?? '',
      data.gcps?.pain_avg ?? '',
      data.gcps?.cpi_0_100 ?? '',
      data.gcps?.interf_daily ?? '',
      data.gcps?.interf_social ?? '',
      data.gcps?.interf_work ?? '',
      data.gcps?.interf_mean_0_100 ?? '',
      data.pain_drawing_count ?? 0,
      Array.isArray(data.pain_drawing_zones) ? data.pain_drawing_zones.join('; ') : '',
      data.jfls8?.mean_0_10 ?? '',
      data.jfls8?.answered_count ?? '',
      JSON.stringify(data.jfls8?.items || {}),
      data.phq4?.total_0_12 ?? '',
      data.phq4?.anxiety_0_6 ?? '',
      data.phq4?.depression_0_6 ?? '',
      data.phq4?.safety_flag ? 'ALERT_REVIEW' : 'NORMAL',
      data.treatment_change || '',
      data.treatment_notes || '',
      data.missing_status || '',
      data.instrument_version || 'DC-TMD-AxisII-v1.2'
    ];

    sheet.appendRow(row);

    return ContentService.createTextOutput(JSON.stringify({
      status: 'success',
      record_uuid: row[0],
      row_number: sheet.getLastRow()
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({
      status: 'error',
      message: err.toString()
    })).setMimeType(ContentService.MimeType.JSON);
  }
}
